/* 指针与数组：一篇讲清 */
#include <stdio.h>

int main(void) {
    int arr[5] = {10, 20, 30, 40, 50};
    int *p = arr;   /* 数组名即首元素地址 */

    for (int i = 0; i < 5; i++)
        printf("arr[%d] = %d\n", i, *(p + i));

    return 0;
}
